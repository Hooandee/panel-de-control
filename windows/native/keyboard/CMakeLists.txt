cmake_minimum_required(VERSION 3.24)
project(PanelKeyboardRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(keyboard-protocol-test tests/protocol_test.cpp)
add_test(NAME keyboard-protocol COMMAND keyboard-protocol-test)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
    message(FATAL_ERROR "The experimental runtime requires MSVC x64")
  endif()
  add_executable(PanelDeControl.KeyboardHost host.cpp)
  add_library(PanelDeControl.KeyboardTap SHARED tap.cpp exports.def)
  foreach(target PanelDeControl.KeyboardHost PanelDeControl.KeyboardTap)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_compile_options(${target} PRIVATE /W4 /permissive- /guard:cf)
    target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
    target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid advapi32)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endforeach()
  install(TARGETS PanelDeControl.KeyboardHost PanelDeControl.KeyboardTap RUNTIME DESTINATION .)
  install(DIRECTORY themes DESTINATION .)
endif()
